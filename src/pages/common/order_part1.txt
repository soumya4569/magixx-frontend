          </button>
        </div>

        <form onSubmit={handleSubmit} className="p-5 space-y-4">
          <div>
            <label className="mb-1 block text-xs font-bold uppercase tracking-wide text-gray-500">Item Name</label>
            <input
              type="text"
              value={name}
              onChange={(e) => setName(e.target.value)}
              placeholder="e.g. Special Extra Cheese"
              className="w-full rounded-xl border border-gray-200 bg-gray-50 px-3.5 py-2.5 text-sm font-semibold text-gray-900 outline-none focus:border-yellow-400 focus:bg-white transition"
              autoFocus
              required

          <div>
            <label className="mb-1 block text-xs font-bold uppercase tracking-wide text-gray-500">Price (â‚¹)</label>

                type="number"
                step="0.5"
                min="0.5"
                value={price}
                onChange={(e) => setPrice(e.target.value)}
